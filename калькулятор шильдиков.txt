    <script>
    (function() {
        let selectedCornerType = 'straight';
        const CORNER_RADIUS = 4; // Фиксированный радиус для всех типов закруглений (4 мм)
        
        // Цвета материалов
        const materialColors = {
            '7': '#F5D76E', // Латунь - более светлый золотистый
            '3': '#C0C0C0', // Сталь - серебристый
            '2.5': '#333333', // Алюминий с черным покрытием - черный
            '4.5': '#FFD700' // Алюминий с золотистым покрытием - золотой
        };

        function calculatePrice() {
            const material = document.getElementById('material').value;
            const materialValue = parseFloat(material);
            const width = parseFloat(document.getElementById('width').value);
            const length = parseFloat(document.getElementById('length').value);
            const lacquer = document.getElementById('lacquer').checked;

            if (isNaN(width) || isNaN(length) || width <= 0 || length <= 0 || width > 200 || length > 200) {
                document.getElementById('result').innerHTML = 'Пожалуйста, введите корректные размеры (от 0 до 200 мм).';
                document.getElementById('visualization').innerHTML = '';
                document.getElementById('total-price').innerHTML = '';
                return;
            }

            const area = (width * length) / 100; // площадь в см²
            let basePrice = 500;
            
            // Обновленная логика базовой стоимости
            if (area < 10) basePrice = 150;
            else if (area >= 10 && area < 15) basePrice = 200;
            else if (area <= 50) basePrice = 300; // Было: else if (area <= 50) basePrice = 300; - это условие перекрывало предыдущие, если area < 15. Исправлено.
            else if (area <= 100) basePrice = 400;
            // Для area > 100 basePrice остается 500 (начальное значение)

            const materialCost = area * materialValue;
            const lacquerCost = lacquer ? area * 1.5 : 0;
            
            // --- НАЧАЛО ИЗМЕНЕНИЙ: Новая логика округления ---
            const unroundedTotal = basePrice + materialCost + lacquerCost;
            let totalPrice; 
            const remainder = unroundedTotal % 50;

            if (remainder >= 10 && remainder <= 20) {
                // Если остаток от 10 до 20 (включительно), округляем ВНИЗ до ближайшего кратного 50
                totalPrice = Math.floor(unroundedTotal / 50) * 50;
            } else if (remainder >= 30 && remainder <= 40) {
                // Если остаток от 30 до 40 (включительно), округляем ВВЕРХ до ближайшего кратного 50
                totalPrice = Math.ceil(unroundedTotal / 50) * 50;
            } else {
                // В остальных случаях (остаток < 10, или между 20 и 30, или > 40)
                // округляем до БЛИЖАЙШЕГО кратного 50
                totalPrice = Math.round(unroundedTotal / 50) * 50;
            }
            // --- КОНЕЦ ИЗМЕНЕНИЙ ---

            document.getElementById('result').innerHTML = `
                <div><strong>Площадь шильдика:</strong> ${area.toFixed(2)} см²</div>
                <div><strong>Базовая стоимость гравировки:</strong> ${basePrice} руб.</div>
                <div><strong>Стоимость материала:</strong> ${materialCost.toFixed(2)} руб.</div>
                ${lacquer ? `<div><strong>Стоимость обработки лаком:</strong> ${lacquerCost.toFixed(2)} руб.</div>` : ''}
            `;

            document.getElementById('total-price').innerHTML = `Общая стоимость: ${totalPrice} руб.`;

            visualizeSize(length, width, material);
        }

        function visualizeSize(length, width, material) {
            const visualizationDiv = document.getElementById('visualization');
            // Максимальный размер визуализации, чтобы она не выходила за пределы контейнера калькулятора
            // Используем offsetWidth родителя (visualization-container) для более точного определения доступной ширины
            const containerWidth = visualizationDiv.parentElement.offsetWidth * 0.9; // 90% ширины контейнера
            const maxSize = Math.min(300, containerWidth); // Не более 300px и не более 90% ширины родителя
            
            // Масштабирование, чтобы большая сторона вписалась в maxSize
            let scale;
            if (length === 0 || width === 0) { // Предотвращение деления на ноль
                 scale = 1;
            } else if (length >= width) {
                scale = maxSize / length;
            } else {
                scale = maxSize / width;
            }
            
            // Если одна из сторон очень маленькая, масштабирование может сделать другую сторону слишком большой
            // Перепроверяем, чтобы обе стороны вписывались
            if (length * scale > maxSize || width * scale > maxSize) {
                 if (length >= width) {
                    scale = maxSize / length;
                } else {
                    scale = maxSize / width;
                }
            }


            const scaledLength = Math.max(1, length * scale); // Минимальный размер 1px для видимости
            const scaledWidth = Math.max(1, width * scale);
            
            visualizationDiv.innerHTML = ''; // Очищаем содержимое
            
            // Устанавливаем размеры контейнера для SVG
            visualizationDiv.style.width = `${scaledLength}px`;
            visualizationDiv.style.height = `${scaledWidth}px`;
            visualizationDiv.style.position = 'relative';
            visualizationDiv.style.overflow = 'visible'; // Для меток размеров
            
            // Добавляем метки размеров, только если размеры не нулевые
            if (length > 0 && width > 0) {
                const widthLabel = document.createElement('div');
                widthLabel.className = 'dimension-label width-label';
                widthLabel.textContent = `${length} мм`;
                
                const lengthLabel = document.createElement('div');
                lengthLabel.className = 'dimension-label length-label';
                lengthLabel.textContent = `${width} мм`;
                
                visualizationDiv.appendChild(widthLabel);
                visualizationDiv.appendChild(lengthLabel);
            }
            
            const svg = document.createElementNS("http://www.w3.org/2000/svg", "svg");
            svg.setAttribute("width", "100%");
            svg.setAttribute("height", "100%");
            svg.style.position = "absolute";
            svg.style.top = "0";
            svg.style.left = "0";
            svg.style.overflow = "visible"; // Чтобы видеть скругления и обводку
            
            const fillColor = materialColors[material] || '#ecf0f1';
            
            const path = document.createElementNS("http://www.w3.org/2000/svg", "path");
            let d = '';
            
            // Масштабированный радиус, но не больше половины меньшей стороны
            const maxPossibleRadius = Math.min(scaledLength, scaledWidth) / 2;
            const scaledRadius = Math.min(CORNER_RADIUS * scale, maxPossibleRadius);
            
            // Убедимся, что длины сторон достаточны для радиуса
            const effectiveLength = Math.max(scaledLength, 2 * scaledRadius);
            const effectiveWidth = Math.max(scaledWidth, 2 * scaledRadius);

            if (selectedCornerType === 'straight' || scaledRadius === 0) {
                d = `
                    M 0 0
                    L ${scaledLength} 0
                    L ${scaledLength} ${scaledWidth}
                    L 0 ${scaledWidth}
                    Z
                `;
            } else if (selectedCornerType === 'rounded') {
                d = `
                    M ${scaledRadius} 0
                    L ${effectiveLength - scaledRadius} 0
                    A ${scaledRadius} ${scaledRadius} 0 0 1 ${effectiveLength} ${scaledRadius}
                    L ${effectiveLength} ${effectiveWidth - scaledRadius}
                    A ${scaledRadius} ${scaledRadius} 0 0 1 ${effectiveLength - scaledRadius} ${effectiveWidth}
                    L ${scaledRadius} ${effectiveWidth}
                    A ${scaledRadius} ${scaledRadius} 0 0 1 0 ${effectiveWidth - scaledRadius}
                    L 0 ${scaledRadius}
                    A ${scaledRadius} ${scaledRadius} 0 0 1 ${scaledRadius} 0
                    Z
                `;
            } else if (selectedCornerType === 'inward') {
                d = `
                    M ${scaledRadius} 0
                    L ${effectiveLength - scaledRadius} 0
                    A ${scaledRadius} ${scaledRadius} 0 0 0 ${effectiveLength} ${scaledRadius} 
                    L ${effectiveLength} ${effectiveWidth - scaledRadius}
                    A ${scaledRadius} ${scaledRadius} 0 0 0 ${effectiveLength - scaledRadius} ${effectiveWidth}
                    L ${scaledRadius} ${effectiveWidth}
                    A ${scaledRadius} ${scaledRadius} 0 0 0 0 ${effectiveWidth - scaledRadius}
                    L 0 ${scaledRadius}
                    A ${scaledRadius} ${scaledRadius} 0 0 0 ${scaledRadius} 0
                    Z
                `;
            }
            
            path.setAttribute("d", d);
            path.setAttribute("fill", fillColor);
            path.setAttribute("stroke", "#3498db");
            path.setAttribute("stroke-width", "2"); // Уменьшил толщину обводки для лучшего вида на малых размерах
            
            svg.appendChild(path);
            visualizationDiv.appendChild(svg);
            
            if (length > 0 && width > 0) {
                const textDiv = document.createElement('div');
                textDiv.textContent = `${length}x${width}мм`; // Более короткая запись
                textDiv.style.position = 'absolute';
                textDiv.style.top = '0';
                textDiv.style.left = '0';
                textDiv.style.width = '100%';
                textDiv.style.height = '100%';
                textDiv.style.display = 'flex';
                textDiv.style.justifyContent = 'center';
                textDiv.style.alignItems = 'center';
                textDiv.style.zIndex = '1';
                textDiv.style.pointerEvents = 'none';
                textDiv.style.fontSize = Math.min(16, scaledWidth / 4, scaledLength / 8) + 'px'; // Адаптивный размер шрифта

                if (material === '2.5') { // Для черного алюминия
                    textDiv.style.color = 'white';
                } else {
                    textDiv.style.color = '#2c3e50'; // Стандартный цвет текста
                }
                visualizationDiv.appendChild(textDiv);
            }
        }

        function selectCornerType(type) {
            selectedCornerType = type;
            
            document.querySelectorAll('.corner-option').forEach(option => {
                option.classList.remove('selected');
            });
            
            document.querySelector(`.corner-option[data-corner-type="${type}"]`).classList.add('selected');
            
            calculatePrice();
        }

        function createInwardCornerPreview() {
            const preview = document.querySelector('.corner-inward .corner-preview');
            preview.innerHTML = ''; 
            
            const svg = document.createElementNS("http://www.w3.org/2000/svg", "svg");
            svg.setAttribute("width", "100%");
            svg.setAttribute("height", "100%");
            
            const path = document.createElementNS("http://www.w3.org/2000/svg", "path");
            
            const width = 58; // Уменьшил, чтобы влезла обводка в 60px контейнер
            const height = 58;
            const radius = 8; 
            
            const d = `
                M ${radius} 0
                L ${width - radius} 0
                A ${radius} ${radius} 0 0 0 ${width} ${radius}
                L ${width} ${height - radius}
                A ${radius} ${radius} 0 0 0 ${width - radius} ${height}
                L ${radius} ${height}
                A ${radius} ${radius} 0 0 0 0 ${height - radius}
                L 0 ${radius}
                A ${radius} ${radius} 0 0 0 ${radius} 0
                Z
            `;
            
            path.setAttribute("d", d);
            path.setAttribute("fill", "#ecf0f1");
            path.setAttribute("stroke", "#3498db");
            path.setAttribute("stroke-width", "2"); // Согласовал толщину обводки
            
            svg.appendChild(path);
            preview.appendChild(svg);
        }

        document.getElementById('material').addEventListener('change', calculatePrice);
        document.getElementById('length').addEventListener('input', calculatePrice);
        document.getElementById('width').addEventListener('input', calculatePrice);
        document.getElementById('lacquer').addEventListener('change', calculatePrice);
        
        document.querySelectorAll('.corner-option').forEach(option => {
            option.addEventListener('click', function() {
                selectCornerType(this.dataset.cornerType);
            });
        });

        window.addEventListener('load', function() {
            createInwardCornerPreview();
            
            document.getElementById('length').value = 100;
            document.getElementById('width').value = 50;
            calculatePrice();
        });

        window.addEventListener('resize', calculatePrice);
    })();
    </script>
